#server.py

import socket
s = socket.socket()
s.bind(("0.0.0.0", 5000))
s.listen(1)
conn, addr = s.accept()
print("Connected:", addr)
while True:
	data = conn.recv(1024).decode()
	if not data or data == "exit":
		break
	print("Client:", data)
	conn.send("Received".encode())
conn.close()


#client.py

import socket
c = socket.socket()
c.connect(("10.10.10.105", 5000)) # Pi's IP
while True:
	msg = input("Enter: ")
	c.send(msg.encode())
	if msg == "exit":
		break
	print("Server:", c.recv(1024).decode())
c.close()
