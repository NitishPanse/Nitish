int redPin = 9;
int greenPin = 10;
int bluePin = 11;

void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);
}

// Function to turn OFF all colours
void clearPins() {
  analogWrite(redPin, 255);
  analogWrite(greenPin, 255);
  analogWrite(bluePin, 255);
}

void loop() {

  // RED
  clearPins();
  analogWrite(redPin, 0);
  delay(1000);

  // GREEN
  clearPins();
  analogWrite(greenPin, 0);
  delay(1000);

  // BLUE
  clearPins();
  analogWrite(bluePin, 0);
  delay(1000);

  // YELLOW
  clearPins();
  analogWrite(redPin, 0);
  analogWrite(greenPin, 0);
  delay(1000);

  // CYAN
  clearPins();
  analogWrite(greenPin, 0);
  analogWrite(bluePin, 0);
  delay(1000);

  // MAGENTA
  clearPins();
  analogWrite(redPin, 0);
  analogWrite(bluePin, 0);
  delay(1000);

  // WHITE
  clearPins();
  analogWrite(redPin, 0);
  analogWrite(greenPin, 0);
  analogWrite(bluePin, 0);
  delay(1000);

  // OFF
  clearPins();
  delay(1000);
}
